This document specifies how to add extra currencies (e.g., seasonal coins) using a scalable, data-driven system.
Seasonal currencies currently **do not expire** and **do not convert**.

## Goals
- Add any number of currencies without adding new DB columns each time
- Keep balances per-user per-currency
- Provide an audit trail (ledger) for debugging/support/anti-cheat
- Ensure **atomic**, **concurrency-safe** spend operations (no negative balances)

## Database Design (Prisma)
Add these models to `schema.prisma`.

```prisma
enum CurrencyType {
  SOFT
  PREMIUM
  SEASONAL
  EVENT
  GUILD
}

model Currency {
  id          String       @id // e.g. "soft_gold", "premium_gems", "season_frost_coin"
  name        String
  symbol      String?
  iconKey     String?
  type        CurrencyType
  seasonId    String?      // optional grouping tag for seasonal currencies
  isTradeable Boolean      @default(false)
  isSpendable Boolean      @default(true)
  decimals    Int          @default(0) // idle games usually 0

  createdAt   DateTime     @default(now())
  updatedAt   DateTime     @updatedAt

  balances    UserCurrencyBalance[]
  ledger      CurrencyLedger[]
}

model UserCurrencyBalance {
  userId     String
  currencyId String
  balance    BigInt      @default(0)
  updatedAt  DateTime    @updatedAt

  user       User        @relation(fields: [userId], references: [id], onDelete: Cascade)
  currency   Currency    @relation(fields: [currencyId], references: [id], onDelete: Restrict)

  @@id([userId, currencyId])
  @@index([currencyId])
  @@index([userId])
}

model CurrencyLedger {
  id         BigInt     @id @default(autoincrement())
  userId     String
  currencyId String

  delta      BigInt
  reason     String     // e.g. "mission_reward", "craft_cost", "shop_purchase"
  sourceType String?    // e.g. "mission", "crafting", "shop", "system", "admin"
  sourceId   String?    // id of mission/shop item/etc
  metadata   Json       @default("{}")
  createdAt  DateTime   @default(now())

  user       User       @relation(fields: [userId], references: [id], onDelete: Cascade)
  currency   Currency   @relation(fields: [currencyId], references: [id], onDelete: Restrict)

  @@index([userId, createdAt(sort: Desc)])
  @@index([currencyId, createdAt(sort: Desc)])
}
